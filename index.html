基本迴圈
for/while/do-while 三種迴圈各有適用場景；範圍 for 是 C++11 的簡潔寫法。
基本迴圈.cpp
#include <iostream> #include <vector> using namespace std; int main() { // === for 迴圈 === for (int i = 0; i < 5; i++) { cout << i << " "; // 0 1 2 3 4 } cout << endl; // 倒數 for (int i = 10; i >= 0; i -= 2) cout << i << " "; // 10 8 6 4 2 0 cout << endl; // 多變數 for (int i = 0, j = 10; i < j; i++, j--) cout << "(" << i << "," << j << ") "; cout << endl; // === while 迴圈 === int n = 1; while (n <= 10) { cout << n << " "; n++; } cout << endl; // === do-while（至少執行一次）=== int input = 0; do { input++; cout << input << " "; } while (input < 5); cout << endl; // === 範圍 for（C++11）=== int arr[] = {10, 20, 30, 40, 50}; for (int x : arr) cout << x << " "; cout << endl; vector<string> words = {"Hello", "C++", "World"}; for (const string &w : words) cout << w << " "; cout << endl; // auto + 參考（可修改元素） for (auto &x : arr) x *= 2; for (auto x : arr) cout << x << " "; // 20 40 60 80 100 return 0; }

算術運算子
算術運算子包含加減乘除、餘數、遞增遞減，注意整數除法會捨去小數部分。
算術運算子.cpp
#include <iostream> #include <cmath> using namespace std; int main() { int a = 17, b = 5; // 基本四則 cout << a + b << endl; // 加: 22 cout << a - b << endl; // 減: 12 cout << a * b << endl; // 乘: 85 cout << a / b << endl; // 整數除法: 3（捨去小數） cout << a % b << endl; // 餘數: 2 // 浮點除法 double x = 17.0, y = 5.0; cout << x / y << endl; // 3.4 cout << (double)a / b << endl; // 3.4（強制轉型） // 遞增 / 遞減 int n = 10; cout << n++ << endl; // 後置++：先用(10)後加 → n=11 cout << n << endl; // 11 cout << ++n << endl; // 前置++：先加後用(12) cout << n-- << endl; // 後置--：先用(12)後減 → n=11 cout << --n << endl; // 前置--：先減後用(10) // 複合賦值 int v = 100; v += 10; cout << v << endl; // 110 v -= 20; cout << v << endl; // 90 v *= 2; cout << v << endl; // 180 v /= 3; cout << v << endl; // 60 v %= 7; cout << v << endl; // 4 // 數學函式 cout << pow(2, 8) << endl; // 256 cout << sqrt(625) << endl; // 25 cout << abs(-42) << endl; // 42 cout << ceil(3.2) << endl; // 4 cout << floor(3.9) << endl; // 3 cout << round(3.5) << endl; // 4 return 0; }

關係與比較運算子
關係運算子回傳布林值；浮點數比較應使用容差而非 ==。
關係與比較運算子.cpp
#include <iostream> #include <string> #include <cmath> using namespace std; int main() { int a = 10, b = 20; // 關係運算子（回傳 bool） cout << (a == b) << endl; // 0：等於 cout << (a != b) << endl; // 1：不等於 cout << (a < b) << endl; // 1：小於 cout << (a > b) << endl; // 0：大於 cout << (a <= b) << endl; // 1：小於等於 cout << (a >= b) << endl; // 0：大於等於 cout << boolalpha; cout << (a < b) << endl; // true // 字串比較 string s1 = "apple", s2 = "banana"; cout << (s1 == s2) << endl; // false cout << (s1 < s2) << endl; // true（字典序） // 浮點數比較（注意精度問題！） double x = 0.1 + 0.2; cout << (x == 0.3) << endl; // false！（浮點誤差） // 正確做法：使用容差比較 double eps = 1e-9; cout << (abs(x - 0.3) < eps) << endl; // true // 找最大值 int scores[] = {85, 92, 78, 96, 81}; int maxScore = scores[0]; for (int i = 1; i < 5; i++) if (scores[i] > maxScore) maxScore = scores[i]; cout << "最高分: " << maxScore << endl; // 96 return 0; }

邏輯運算子
&& 邏輯且、|| 邏輯或、! 邏輯非；短路求值可避免不必要的計算或錯誤。
邏輯運算子.cpp
#include <iostream> using namespace std; int main() { bool a = true, b = false; cout << boolalpha; cout << (a && b) << endl; // AND: false cout << (a || b) << endl; // OR: true cout << (!a) << endl; // NOT: false cout << (!b) << endl; // NOT: true // 真值表 bool T = true, F = false; cout << "T&&T=" << (T&&T) << " T&&F=" << (T&&F) << endl; cout << "T||F=" << (T||F) << " F||F=" << (F||F) << endl; // 短路求值（Short-circuit evaluation） int x = 0; // && 左側為 false，右側不執行（避免除以零） if (x != 0 && 10/x > 1) cout << "安全" << endl; else cout << "x 為零，已跳過除法" << endl; // 實用範例 int age = 20; bool hasID = true; if (age >= 18 && hasID) cout << "可入場" << endl; int score = 75; bool hasBonus = false; if (score >= 90 || hasBonus) cout << "獲得獎勵" << endl; else cout << "未獲獎勵" << endl; // 複合條件 int n = 55; if (n > 0 && n < 100 && n % 5 == 0) cout << n << " 是 1~99 的 5 的倍數" << endl; // 德摩根定理：!(A && B) == !A || !B bool p = true, q = false; cout << !(p && q) << endl; // true cout << (!p || !q) << endl; // true（等價） return 0; }

位元運算子
位元運算直接操作二進位，效率高，常用於旗標、加密、圖形處理。
位元運算子.cpp
#include <iostream> #include <bitset> using namespace std; int main() { int a = 0b1010; // 10（二進位 1010） int b = 0b1100; // 12（二進位 1100） cout << bitset<8>(a) << endl; // 00001010 cout << bitset<8>(b) << endl; // 00001100 // 位元 AND：兩位皆1才1 cout << (a & b) << " = " << bitset<8>(a&b) << endl; // 8 // 位元 OR：任一為1則1 cout << (a | b) << " = " << bitset<8>(a|b) << endl; // 14 // 位元 XOR：相同為0，不同為1 cout << (a ^ b) << " = " << bitset<8>(a^b) << endl; // 6 // 位元 NOT：取反 cout << (~a) << endl; // -11 // 左移（×2^n） cout << (a << 1) << endl; // 20 cout << (a << 2) << endl; // 40 // 右移（÷2^n） cout << (b >> 1) << endl; // 6 cout << (b >> 2) << endl; // 3 // 位元技巧 int n = 42; cout << ((n & 1) ? "奇數" : "偶數") << endl; // 偶數 int k = 3; n |= (1 << k); // 設定第3位 n &= ~(1 << k); // 清除第3位 n ^= (1 << k); // 翻轉第3位 cout << (100 << 3) << endl; // 100*8 = 800 cout << (100 >> 2) << endl; // 100/4 = 25 return 0; }

#include <iostream> #include <tuple> using namespace std; int add(int a, int b) { return a + b; } double average(int arr[], int n) { int sum = 0; for (int i = 0; i < n; i++) sum += arr[i]; return (double)sum / n; } string classify(int score) { if (score >= 90) return "優秀"; if (score >= 70) return "良好"; if (score >= 60) return "及格"; return "不及格"; } // 回傳多個值（tuple） tuple<int, int> divmod(int a, int b) { return {a / b, a % b}; } int main() { cout << add(3, 4) << endl; // 7 int scores[] = {80, 90, 75, 85}; cout << average(scores, 4) << endl; // 82.5 cout << classify(88) << endl; // 良好 auto [q, r] = divmod(17, 5); cout << "商=" << q << " 餘=" << r << endl; return 0; }

#include <iostream> using namespace std; // 階乘 int factorial(int n) { if (n <= 1) return 1; return n * factorial(n - 1); } // 費波那契數列 int fib(int n) { if (n <= 1) return n; return fib(n-1) + fib(n-2); } // 河內塔 void hanoi(int n, char from, char to, char aux) { if (n == 1) { cout << "移動圓盤 1: " << from << " -> " << to << endl; return; } hanoi(n-1, from, aux, to); cout << "移動圓盤 " << n << ": " << from << " -> " << to << endl; hanoi(n-1, aux, to, from); } // 二分搜尋（遞迴版） int binarySearch(int arr[], int low, int high, int target) { if (low > high) return -1; int mid = (low + high) / 2; if (arr[mid] == target) return mid; if (arr[mid] > target) return binarySearch(arr, low, mid-1, target); return binarySearch(arr, mid+1, high, target); } int main() { cout << factorial(6) << endl; // 720 cout << fib(10) << endl; // 55 hanoi(3, 'A', 'C', 'B'); int arr[] = {1, 3, 5, 7, 9, 11, 13}; cout << binarySearch(arr, 0, 6, 7) << endl; // 3 return 0; }

#include <iostream> #include <string> using namespace std; class Circle { public: const double PI = 3.14159265358979; const int ID; double radius; Circle(int id, double r) : ID(id), radius(r) {} double area() const { return PI * radius * radius; } }; class Config { public: const string APP_NAME; const int VERSION; const double MAX_LOAD; Config(string name, int ver, double load) : APP_NAME(name), VERSION(ver), MAX_LOAD(load) {} void show() const { cout << APP_NAME << " v" << VERSION << " (max: " << MAX_LOAD << ")" << endl; } }; class MathConst { public: static constexpr double PI = 3.14159265358979; static constexpr double E = 2.71828182845904; static constexpr int MAX = 1000000; static double circleArea(double r) { return PI * r * r; } }; int main() { Circle c1(1, 5.0), c2(2, 3.0); cout << "C1 面積: " << c1.area() << endl; cout << "C2 面積: " << c2.area() << endl; Config cfg("MyApp", 2, 0.85); cfg.show(); cout << MathConst::PI << endl; cout << MathConst::circleArea(10) << endl; return 0; }

靜態資料成員
靜態成員屬於類別本身而非個別物件，所有物件共用，用 ClassName:: 存取。
靜態資料成員.cpp
#include <iostream> #include <string> using namespace std; class Student { private: string name; int id; static int count; static int nextId; static string school; public: Student(string n) : name(n), id(++nextId) { count++; cout << name << " 已加入，共 " << count << " 人" << endl; } ~Student() { count--; cout << name << " 離開，剩 " << count << " 人" << endl; } static int getCount() { return count; } static void setSchool(string s) { school = s; } static void showSchool() { cout << "學校: " << school << " 人數: " << count << endl; } void info() const { cout << school << " | " << id << " | " << name << endl; } }; int Student::count = 0; int Student::nextId = 1000; string Student::school = "C++ 大學"; int main() { cout << "學生數: " << Student::getCount() << endl; Student::setSchool("國立程式大學"); { Student s1("Alice"); Student s2("Bob"); Student s3("Charlie"); s1.info(); Student::showSchool(); } cout << "最終: " << Student::getCount() << " 人" << endl; return 0; }
